# Base packages — shared by both headless and desktop variants
# These come from the core LFS build (Stage 1-3) and are common to
# any bootable system regardless of GUI/headless split.

linux-kernel         # custom-configured per config/kernel-notes.md
musl                  # used across both variants
apk-tools              # package manager — shared across both variants
bash
coreutils
util-linux
grub                 # BIOS + UEFI boot support (see 07-build-iso.sh)
busybox              # optional: can replace many coreutils for smaller footprint
openrc               # init system
iproute2
openssh              # remote management on headless; optional on desktop
nano                 # default editor
